	t.Logf("Processing 503 before recovery; identical TaskObservation, deliveries=%d applications=%d outbox=ACKED", sent, applied)

func runtimeIntent(t *testing.T) ([]byte, intake.Intent, intake.Snapshot) {
	t.Helper()
	s := intake.Snapshot{
		Workspace: uuid.New(), Case: uuid.New(), Operation: uuid.New(), Config: uuid.New(),
		Lineage: uuid.New(), Episode: uuid.New(), Connection: uuid.New(), SourceFacts: uuid.New(),
		Epoch: 1, ConfigDigest: "sha256:" + strings.Repeat("a", 64), FactsDigest: "sha256:" + strings.Repeat("b", 64),
		FactsUntil: time.Now().UTC().Truncate(time.Microsecond).Add(2 * time.Minute), Deadline: time.Now().UTC().Truncate(time.Microsecond).Add(3 * time.Minute),
		DestinationWorkspace: "fixture-workspace", List: "456", Assignee: exactAssignee,
	}
	fields := map[string]any{
		"schema_version": "1.0", "rendering_version": "handoff-block-v1", "connection_epoch": s.Epoch,
		"config_digest": s.ConfigDigest, "source_facts_digest": s.FactsDigest, "facts_valid_until": s.FactsUntil, "deadline_at": s.Deadline,
		"destination_workspace_id": s.DestinationWorkspace, "target_list_id": s.List, "assignee_id": s.Assignee,
		"title": fixtureTitle, "service_label": "Example", "source_session_id": "fixture-session", "order_reference": "order",
		"case_marker": "fixture-case", "rendered_handoff_block": canonicalTestBlock(s.Case), "context": []any{},
		"message_id": uuid.New(), "workspace_id": s.Workspace, "case_id": s.Case, "operation_id": s.Operation,
		"config_id": s.Config, "lineage_id": s.Lineage, "episode_id": s.Episode, "connection_id": s.Connection, "source_facts_id": s.SourceFacts,
	}
	raw, err := json.Marshal(fields)
	if err != nil {
		t.Fatal(err)
	}
	canonical, err := jcs.Transform(raw)
	if err != nil {
		t.Fatal(err)
	}
	fields["message_digest"] = checksum(canonical)
	raw, err = json.Marshal(fields)
	if err != nil {
		t.Fatal(err)
	}
	intent, err := intake.DecodeIntent(raw)
	if err != nil {
		t.Fatal(err)
	}
	return raw, intent, s
}

func runtimeHTTP(t *testing.T, ca *testCerts, cert tls.Certificate) *http.Client {
	t.Helper()
	c := &http.Client{Timeout: 2 * time.Second, Transport: &http.Transport{
		TLSClientConfig:   &tls.Config{MinVersion: tls.VersionTLS13, RootCAs: ca.roots, Certificates: []tls.Certificate{cert}},

func runtimeRequest(c *http.Client, method, url string, body []byte) (int, []byte, error) {
	req, err := http.NewRequest(method, url, bytes.NewReader(body))
	if err != nil {
		return 0, nil, err
	}
	req.Header.Set("X-Request-ID", uuid.NewString())
	req.Header.Set("Content-Type", "application/json")
	res, err := c.Do(req)
	if err != nil {
		return 0, nil, err
	}
	defer res.Body.Close()
	raw, err := io.ReadAll(io.LimitReader(res.Body, 65537))
	return res.StatusCode, raw, err
}

func runtimeEnv(env, overrides map[string]string) []string {
	all := os.Environ()
	for k, v := range env {
		if replacement, ok := overrides[k]; ok {
			v = replacement
		}
		all = append(all, k+"="+v)
	}
	return all
}

func refuses(t *testing.T, path string, env, overrides map[string]string, reason string) {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 6*time.Second)
	defer cancel()
	cmd := exec.CommandContext(ctx, path)
	cmd.Env = runtimeEnv(env, overrides)
	raw, err := cmd.CombinedOutput()
	exit, ok := err.(*exec.ExitError)
	if !ok || exit.ExitCode() != 1 || !strings.Contains(string(raw), reason) {
		t.Fatalf("expected closed startup (%s), err=%v output=%s", reason, err, raw)
	}
	t.Logf("closed startup exit=1 (%s)", reason)
}

func runtimeStart(t *testing.T, path string, env map[string]string) func() {
	t.Helper()
	cmd := exec.Command(path)
	cmd.Env = runtimeEnv(env, nil)
	var output bytes.Buffer
	cmd.Stdout, cmd.Stderr = &output, &output
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	stop := sync.OnceFunc(func() {
		_ = cmd.Process.Kill()
		_ = cmd.Wait()
		t.Logf("Delivery process log: %s", strings.TrimSpace(output.String()))
	})
	t.Cleanup(stop)
	return stop
}

func runtimeEventually(t *testing.T, want string, match func() bool) {
	t.Helper()
	until := time.Now().Add(14 * time.Second)
	for time.Now().Before(until) {
		if match() {
			return
		}
		time.Sleep(50 * time.Millisecond)
	}
	t.Fatalf("not observed: %s", want)
}

type testCerts struct {
	cert  *x509.Certificate
	key   *ecdsa.PrivateKey
	roots *x509.CertPool
}

func newReadCA(t *testing.T) *testCerts {
	t.Helper()
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		t.Fatal(err)
	}
	tmpl := certTemplate()
	tmpl.IsCA, tmpl.BasicConstraintsValid = true, true
	tmpl.KeyUsage = x509.KeyUsageCertSign
	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if err != nil {
		t.Fatal(err)
	}
	cert, err := x509.ParseCertificate(der)
	if err != nil {
		t.Fatal(err)
	}
	roots := x509.NewCertPool()
	roots.AddCert(cert)
	return &testCerts{cert, key, roots}
}
func certTemplate() *x509.Certificate {
	return &x509.Certificate{SerialNumber: big.NewInt(time.Now().UnixNano()), NotBefore: time.Now().Add(-time.Hour),
		NotAfter: time.Now().Add(time.Hour), KeyUsage: x509.KeyUsageDigitalSignature}
}
func (ca *testCerts) issue(t *testing.T, san string, usage x509.ExtKeyUsage) tls.Certificate {
	t.Helper()
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		t.Fatal(err)
	}
	tmpl := certTemplate()
	tmpl.ExtKeyUsage = []x509.ExtKeyUsage{usage}
	if san == "127.0.0.1" {
		tmpl.IPAddresses = append(tmpl.IPAddresses, []byte{127, 0, 0, 1})
	} else {
		tmpl.DNSNames = []string{san}
	}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, ca.cert, &key.PublicKey, ca.key)
	if err != nil {
		t.Fatal(err)
	}
	return tls.Certificate{Certificate: [][]byte{der}, PrivateKey: key}
}

}

			RootCAs: ca.roots, MinVersion: tls.VersionTLS13, Certificates: []tls.Certificate{cert},

		t.Fatal(err)

		t.Fatalf("Processing rate limit: %v", err)

		FROM hp_delivery.outbox o JOIN hp_delivery.job_index j
		ON j.workspace_id=o.workspace_id AND j.job_id=o.message_id
		WHERE o.message_id=$1 AND o.state='PENDING'`, o.MessageID).Scan(&nextAttempt, &indexedDue)
	if err != nil || nextAttempt.Before(beforeRateLimit.Add(6*time.Minute-time.Microsecond)) || !nextAttempt.Equal(indexedDue) {
		t.Fatalf("Retry-After not persisted to outbox and job index: %v, %v, %v", nextAttempt, indexedDue, err)
	}
	if err := restart.Publish(t.Context(), o.Workspace, o.MessageID, q.Epoch); !errors.Is(err, intake.ErrNotFound) || deliveries.Load() != 2 {
		t.Fatalf("premature delivery during Retry-After: %v, posts=%d", err, deliveries.Load())
	}
	makeDue()
	if err = restart.Publish(t.Context(), o.Workspace, o.MessageID, q.Epoch); err != nil {
		t.Fatal(err)
	}
	var state string
	err = f.admin.QueryRow(t.Context(), "SELECT state FROM hp_delivery.outbox WHERE message_id=$1", o.MessageID).Scan(&state)
	if err != nil || state != "ACKED" || deliveries.Load() != 3 {
		t.Fatalf("outbox replay incorrect: %s %v", state, err)
	}
	next, err := service.QueryKnownTask(t.Context(), q)
	if err != nil || next.Observation == o.Observation || reads.Load() != 2 || next.EvidenceExpires == nil ||
		!next.EvidenceExpires.Equal(*o.EvidenceExpires) {
		t.Fatal("new observation changed identity or retention")
	}
	claim, err := reopened.claimObservation(t.Context(), next.Workspace, next.MessageID)
	if err != nil {
		t.Fatal(err)
	}
	if err := reopened.finishPublication(t.Context(), next.Workspace, claim, false, &claim.ref.Expires); err != nil {
		t.Fatal(err)
	}
	var held, stillScheduled bool
	err = f.admin.QueryRow(t.Context(), `SELECT o.state='HELD',
		EXISTS (SELECT 1 FROM hp_delivery.job_index j WHERE j.workspace_id=o.workspace_id AND j.job_id=o.message_id)
		FROM hp_delivery.outbox o WHERE o.message_id=$1`, next.MessageID).Scan(&held, &stillScheduled)
	if err != nil || !held || stillScheduled {
		t.Fatalf("expired retry window must hold without scheduling: held=%t scheduled=%t err=%v", held, stillScheduled, err)
	}
	t.Logf("native ID=%s knowledge=%s lineage=%s fence=%t; Processing deliveries=%d (same message/digest), Retry-After=360s persisted; outbox=%s", id, o.Knowledge, observedState, fence, deliveries.Load(), state)
}

func TestReadbackFailures(t *testing.T) {
	f := setup(t)
	fp := f.budget(t, 50)
	_, a, q := knownFixture(t, f, fp)
	for _, tc := range []string{"mismatch", "content", "incomplete", "wrong-id", "403", "404", "429", "malformed", "oversize", "timeout", "denied", "new-epoch", "wrong-target", "cancel", "wrong-ca", "budget", "expired-file", "corrupt-file", "lost-file"} {
		t.Run(tc, func(t *testing.T) {
			var gets atomic.Int64
			target := fakeClickUp(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				gets.Add(1)
				if r.Method != http.MethodGet {
					t.Error("a provider write was attempted")
				}
				switch tc {
				case "403", "404", "429":
					var n int
					fmt.Sscan(tc, &n)
					w.WriteHeader(n)
				case "malformed":
					io.WriteString(w, "{")
				case "oversize":
					w.Write(bytes.Repeat([]byte("x"), maxResponse+1))
				case "timeout":
					select {
					case <-r.Context().Done():
					case <-time.After(5 * time.Second):
					}
				default:
					w.Write(targetJSON(t, a, "fixture-task", tc))
				}
			}))
			service := queryFixture(t, f, a, fp, target)
			query := q
			ctx := t.Context()
			var wantErr error
			switch tc {
			case "denied":
				service.authority = readAuth(func(context.Context, intake.Query) (Authorization, error) { return Authorization{}, intake.ErrDenied })
				wantErr = intake.ErrDenied
			case "new-epoch":
				query.Epoch++
				wantErr = intake.ErrDenied
			case "wrong-target":
				query.TaskID = "different-task"
				wantErr = intake.ErrConflict
			case "cancel":
				c, cancel := context.WithCancel(ctx)
				cancel()
				ctx = c
				wantErr = intake.ErrUnavailable
			case "wrong-ca":
				target.tlsConfig.RootCAs = x509.NewCertPool()
				wantErr = intake.ErrUnavailable
			case "budget":
				if _, err := f.admin.Exec(ctx, "UPDATE hp_delivery.credential_budget SET limit_count=used_count WHERE token_fingerprint=$1", fp[:]); err != nil {
					t.Fatal(err)
				}
				t.Cleanup(func() {
					_, _ = f.admin.Exec(context.Background(), "UPDATE hp_delivery.credential_budget SET limit_count=50 WHERE token_fingerprint=$1", fp[:])
				})
				wantErr = intake.ErrQuota
			case "expired-file", "corrupt-file":
				k, err := f.worker.known(ctx, q, a)
				if err != nil {
					t.Fatal(err)
				}
				if tc == "expired-file" {
					if _, err := f.admin.Exec(ctx, "UPDATE hp_delivery.intent_content SET expires_at=clock_timestamp()-interval '1 minute' WHERE attempt_id=$1", a.Attempt); err != nil {
						t.Fatal(err)
					}
					t.Cleanup(func() {
						if _, err := f.admin.Exec(context.Background(), "UPDATE hp_delivery.intent_content SET expires_at=$2 WHERE attempt_id=$1", a.Attempt, k.ref.Expires); err != nil {
							t.Error(err)
						}
					})
				} else {
					filename := filepath.Join(f.root, k.ref.ID.String())
					original, err := os.ReadFile(filename)
					if err != nil {
						t.Fatal(err)
					}
					if err = os.WriteFile(filename, []byte("corrupt"), 0600); err != nil {
						t.Fatal(err)
					}
					t.Cleanup(func() {
						if err := os.WriteFile(filename, original, 0600); err != nil {
							t.Error(err)
						}
					})
				}
				wantErr = intake.ErrUnavailable
			case "lost-file":
				k, err := f.worker.known(ctx, q, a)
				if err != nil {
					t.Fatal(err)
				}
				if err = os.Remove(filepath.Join(f.root, k.ref.ID.String())); err != nil {
					t.Fatal(err)
				}
				wantErr = intake.ErrUnavailable
			}
			o, err := service.QueryKnownTask(ctx, query)
			if wantErr != nil {
				if !errors.Is(err, wantErr) || gets.Load() != 0 {
					t.Fatalf("bad refusal: %v, GET count %d", err, gets.Load())
				}
				return
			}
			want := "INCOMPLETE"
			if tc == "mismatch" || tc == "content" {
				want = "RESULT_MISMATCH"
			}
			if err != nil || o.Knowledge != want || gets.Load() != 1 {
				t.Fatalf("expected %s received %+v %v", want, o, err)
			}
			if tc == "incomplete" && o.WorkspaceMatches != nil {
				t.Fatal("missing provider workspace was substituted")
			}
			if tc == "wrong-id" && o.EvidenceRef != nil {
				t.Fatal("foreign identity linked as evidence")
			}
		})
	}
}

type allowGate func(context.Context, uuid.UUID, int64, func() error) error

func (g allowGate) WithOpen(ctx context.Context, w uuid.UUID, e int64, f func() error) error {
	return g(ctx, w, e, f)
}

type credentialSource func(context.Context, uuid.UUID, uuid.UUID, int64) (Credential, error)

func (f credentialSource) ForConnection(ctx context.Context, w, c uuid.UUID, e int64) (Credential, error) {
	return f(ctx, w, c, e)
}

	app    *intake.Postgres
	worker *Postgres
	dir    *intake.Directory
	root   string

func connect(t *testing.T, env string) *pgxpool.Pool {
	t.Helper()
	p, err := pgxpool.New(t.Context(), os.Getenv(env))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(p.Close)
	return p
}

func setup(t *testing.T) *fixture {
	t.Helper()
	if os.Getenv("HIT92_TEST_WORKER_DSN") == "" {
		if os.Getenv("HIT92_REQUIRE_POSTGRES") == "1" {
			t.Fatal("worker fixture not configured")
		}
		t.Skip("isolated worker fixture not requested")
	}
	admin := connect(t, "HIT91_TEST_ADMIN_DSN")
	var version int
	if err := admin.QueryRow(t.Context(), "SELECT current_setting('server_version_num')::int").Scan(&version); err != nil || version != 180007 {
		t.Fatalf("expected PostgreSQL 18.7: %d %v", version, err)
	}
	root := filepath.Join(t.TempDir(), "content")
	if err := os.Mkdir(root, 0700); err != nil {
		t.Fatal(err)
	}
	dir, err := intake.OpenDirectory(root)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = dir.Close() })
	app, err := intake.NewPostgres(connect(t, "HIT91_TEST_APP_DSN"), dir)
	if err != nil {
		t.Fatal(err)
	}
	worker, err := NewPostgres(connect(t, "HIT92_TEST_WORKER_DSN"))
	if err != nil {
		t.Fatal(err)
	}
	return &fixture{admin, app, worker, dir, root}
}

func TestPostgresSuccessAndSharedBudget(t *testing.T) {
	f := setup(t)
	fp := f.budget(t, 1)
	job, a := f.prepare(t, fp, false)
	var sends atomic.Int64
	client := fakeClickUp(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		sends.Add(1)
		body, err := io.ReadAll(r.Body)
		var got struct {
			Name        string
			Description string
			Assignees   []json.Number
		}
		if err != nil || json.Unmarshal(body, &got) != nil || r.URL.Path != "/api/v2/list/456/task" ||
			r.Method != http.MethodPost || r.Host == "" || r.ProtoMajor != 1 || !r.Close ||
			r.Header.Get("Authorization") != fixtureToken || r.Header.Get("Idempotency-Key") != "" ||
			got.Name != fixtureTitle || got.Description != fixtureBlock || len(got.Assignees) != 1 || got.Assignees[0] != exactAssignee {
			t.Error("unexpected HTTPS request")
		}
		var state string
		var fenced, started bool
		tx, err := f.admin.Begin(r.Context())
		if err != nil {
			t.Error("read fence: ", err)
			return
		}
		defer tx.Rollback(context.Background())
		err = tx.QueryRow(r.Context(), `SELECT l.state,l.ever_dispatch_fenced,a.network_start_at IS NOT NULL
			FROM hp_delivery.effect_lineage l JOIN hp_delivery.attempts a USING(workspace_id,lineage_id)
			WHERE a.workspace_id=$1 AND a.attempt_id=$2 AND l.lineage_id=$3 FOR UPDATE OF l,a NOWAIT`,
			a.Workspace, job.Attempt, a.Lineage).Scan(&state, &fenced, &started)
		if err != nil || state != "DISPATCHING" || !fenced || !started {
			t.Errorf("fence or row lock at HTTP boundary: %s %t %t %v", state, fenced, started, err)
		}
		io.WriteString(w, `{"id":"opaque-test-task"}`)
	}))
	worker := f.newWorker(t, client, a, fp)
	var wg sync.WaitGroup
	results := make(chan Result, 8)
	for i := 0; i < 8; i++ {
		wg.Go(func() {
			got, err := worker.Run(t.Context(), job)
			if err == nil {
				results <- got
			} else if !errors.Is(err, ErrFenced) {
				t.Errorf("unexpected concurrent result: %v", err)
			}
		})
	}
	wg.Wait()
	close(results)
	var count int
	for x := range results {
		count++
		if x.Disposition != TargetIDObserved || x.TaskID != "opaque-test-task" {
			t.Errorf("unexpected outcome: %+v", x)
		}
	}
	state, knowledge, id, fenced, started := f.state(t, job)
	if sends.Load() != 1 || count != 1 || state != "RESULT_ID_OBSERVED" || knowledge != "TARGET_ID_OBSERVED" ||
		id != "opaque-test-task" || !fenced || !started {
		t.Fatal("one ID-only observation expected")
	}
	other, b := f.prepare(t, fp, false)
	second := f.newWorker(t, client, b, fp)
	result, err := second.Run(t.Context(), other)
	if err != nil || result.Disposition != Ambiguous || sends.Load() != 1 {
		t.Fatal("shared budget did not hold the second workspace")
	}
	var used int
	if err := f.admin.QueryRow(t.Context(), "SELECT used_count FROM hp_delivery.credential_budget WHERE token_fingerprint=$1", fp[:]).Scan(&used); err != nil || used != 1 {
		t.Fatal("shared budget was charged more than once")
	}
	t.Logf("HTTPS sends=%d; state=%s; knowledge=%s; opaque task ID=%s; shared budget used=%d", sends.Load(), state, knowledge, id, used)
}

func TestPostgresUnknownAndDenied(t *testing.T) {
	f := setup(t)
	for _, name := range []string{"redirect", "401", "403", "429", "503", "malformed", "oversize", "abort", "timeout", "gate", "epoch", "revoked", "expired-facts", "deadline", "cancelled", "credential", "lost-content", "legacy", "crash"} {
		t.Run(name, func(t *testing.T) {
			fp := f.budget(t, 1)
			job, a := f.prepare(t, fp, name == "legacy")
			var sends atomic.Int64
			client := fakeClickUp(t, http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				sends.Add(1)
				switch name {
				case "redirect":
					w.Header().Set("Location", "https://unreachable.invalid/redirected")
					w.WriteHeader(307)
				case "401", "403", "429", "503":
					var status int
					fmt.Sscan(name, &status)
					w.WriteHeader(status)
				case "malformed":
					io.WriteString(w, `{"id":`)
				case "oversize":
					w.Write(bytes.Repeat([]byte(" "), maxResponse+1))
				case "abort":
					panic(http.ErrAbortHandler)
				case "timeout":
					_, _ = io.Copy(io.Discard, r.Body)
					select {
					case <-r.Context().Done():
					case <-time.After(9 * time.Second):
					}
				default:
					t.Error("unexpected provider request")
				}
			}))
			w := f.newWorker(t, client, a, fp)
			ctx := t.Context()
			switch name {
			case "gate":
				w.gate = allowGate(func(context.Context, uuid.UUID, int64, func() error) error { return ErrHeld })
			case "epoch":
				if _, err := f.admin.Exec(ctx, "UPDATE hp_delivery.destination_connections SET security_epoch=2 WHERE workspace_id=$1", job.Workspace); err != nil {
					t.Fatal(err)
				}
			case "revoked":
				if _, err := f.admin.Exec(ctx, "UPDATE hp_delivery.destination_connections SET revoked_at=clock_timestamp() WHERE workspace_id=$1", job.Workspace); err != nil {
					t.Fatal(err)
				}
			case "expired-facts", "deadline":
				if name == "deadline" {
					a.Deadline = time.Now().Add(-time.Minute)
				} else {
					a.FactsUntil = time.Now().Add(-time.Minute)
				}
				w = f.newWorker(t, client, a, fp)
			case "cancelled":
				var cancel context.CancelFunc
				ctx, cancel = context.WithCancel(ctx)
				cancel()
			case "credential":
				w.credentials = credentialSource(func(context.Context, uuid.UUID, uuid.UUID, int64) (Credential, error) { return Credential{}, ErrHeld })
			case "lost-content":
				rec, err := f.worker.load(ctx, job)
				if err != nil {
					t.Fatal(err)
				}
				if err := os.Remove(filepath.Join(f.root, rec.ref.ID.String())); err != nil {
					t.Fatal(err)
				}
			case "crash":
				fenceInCrashedProcess(t, job)
			}
			result, err := w.Run(ctx, job)
			state, knowledge, id, fenced, started := f.state(t, job)
			switch name {
			case "redirect", "401", "403", "429", "503", "malformed", "oversize", "abort", "timeout":
				if err != nil || result.Disposition != Ambiguous || sends.Load() != 1 || state != "AMBIGUOUS" ||
					knowledge != "POSSIBLE_EFFECT" || id != "" || !fenced || !started {
					t.Fatalf("not a fenced unknown outcome: %v %s %s", err, state, knowledge)
				}
				restart := f.newWorker(t, client, a, fp)
				if _, err := restart.Run(t.Context(), job); !errors.Is(err, ErrFenced) || sends.Load() != 1 {
					t.Fatal("restart repeated create")
				}
				if name == "abort" {
					t.Logf("lost response: state=%s, knowledge=%s, initial sends=1, repeated sends=0", state, knowledge)
				}
			case "crash":
				if !errors.Is(err, ErrFenced) || sends.Load() != 0 || state != "DISPATCHING" || !fenced || started {
					t.Fatal("crash lost its durable fence")
				}
				t.Log("process exited after SQL fence; restarted worker sent 0 requests")
			default:
				if !errors.Is(err, ErrHeld) || sends.Load() != 0 || state != "PREPARED" || fenced || started {
					t.Fatalf("protected request was not held: %v %s", err, state)
				}
			}
		})
	}
}

func fenceInCrashedProcess(t *testing.T, job Job) {
	t.Helper()
	binary, err := os.Executable()
	if err != nil {
		t.Fatal(err)
	}
	cmd := exec.CommandContext(t.Context(), binary, "-test.run=^TestFenceCrashProcess$")
	cmd.Env = append(os.Environ(), "HIT92_CRASH_WORKSPACE="+job.Workspace.String(), "HIT92_CRASH_ATTEMPT="+job.Attempt.String())
	out, err := cmd.CombinedOutput()
	var exit *exec.ExitError
	if !errors.As(err, &exit) || exit.ExitCode() != 42 {
		t.Fatalf("fence process failed: %v %s", err, out)
	}
}

func TestFenceCrashProcess(t *testing.T) {
	workspace := os.Getenv("HIT92_CRASH_WORKSPACE")
	if workspace == "" {
		t.Skip("used by the restart fixture")
	}
	job := Job{uuid.MustParse(workspace), uuid.MustParse(os.Getenv("HIT92_CRASH_ATTEMPT"))}
	db, err := NewPostgres(connect(t, "HIT92_TEST_WORKER_DSN"))
	if err != nil {
		t.Fatal(err)
	}
	rec, err := db.load(t.Context(), job)
	if err != nil {
		t.Fatal(err)
	}
	if err = db.fence(t.Context(), rec); err != nil {
		t.Fatal(err)
	}
	os.Exit(42)
}
